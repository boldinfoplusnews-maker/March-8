    <script>
        function toggleMusic() {
            var audio = document.getElementById("myAudio");
            var btn = document.getElementById("playBtn");

            if (audio.paused) {
                audio.play();
                btn.innerHTML = "⏸ Pause Music";
                btn.style.backgroundColor = "#ea4335"; // Changes to red when playing
            } else {
                audio.pause();
                btn.innerHTML = "🎵 Play Music";
                btn.style.backgroundColor = "#4285f4"; // Back to blue when paused
            }
        }
    </script>
